
/*
========================================================
 PROJECT: KEYPAD 4x4 + LCD + WIFI + MQTT
 BOARD: Arduino UNO R4 WiFi
========================================================

 อุปกรณ์:
 1. Arduino UNO R4 WiFi
 2. Keypad Matrix 4x4
 3. LCD I2C 16x2
 4. สาย Jumper และ Breadboard

 การต่อ Keypad:
 R1 -> D2
 R2 -> D3
 R3 -> D4
 R4 -> D5
 C1 -> D6
 C2 -> D7
 C3 -> D8
 C4 -> D9

 การต่อ LCD I2C:
 GND -> Arduino GND
 VCC -> Arduino 5V
 SDA -> Arduino SDA
 SCL -> Arduino SCL

 Libraries:
 - WiFiS3
 - MQTT by Joel Gaehwiler (256dpi)
 - Wire
 - LiquidCrystal_I2C

 ไม่ใช้ Keypad.h
 อ่าน Keypad ด้วย digitalRead/digitalWrite

 การทำงาน:
 1. เชื่อมต่อ Wi-Fi
 2. เชื่อมต่อ MQTT Broker
 3. สแกนปุ่ม Keypad 4x4 โดยตรง
 4. กรองการเด้งของปุ่ม (Debounce)
 5. กด 1 ครั้ง ส่ง MQTT 1 ครั้ง
 6. แสดงปุ่มล่าสุดบน LCD
 7. แสดงสถานะ MQTT บน LCD
 8. เชื่อมต่อใหม่อัตโนมัติเมื่อหลุด

 MQTT Broker: phycom.it.kmitl.ac.th
 MQTT Port: 1883
 MQTT Topic: 68070131/keypad

 Serial Monitor: 9600 baud
 LCD I2C Address: 0x27
========================================================
*/

#include <WiFiS3.h>
#include <MQTTClient.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// ====================================================
// 1. WIFI CONFIGURATION
// ====================================================

// ชื่อ Wi-Fi
const char WIFI_SSID[] = "EDOK";

// รหัสผ่าน Wi-Fi
const char WIFI_PASSWORD[] = "1234567890";

// ====================================================
// 2. MQTT CONFIGURATION
// ====================================================

// MQTT Broker ของรายวิชา
const char MQTT_BROKER[] = "phycom.it.kmitl.ac.th";

// พอร์ต MQTT
const int MQTT_PORT = 1883;

// Topic สำหรับส่งปุ่ม Keypad
const char MQTT_TOPIC[] = "68070131/keypad";

// สร้าง TCP Client
WiFiClient network;

// สร้าง MQTT Client
MQTTClient mqtt(256);

// ====================================================
// 3. LCD CONFIGURATION
// ====================================================

// LCD I2C Address 0x27 ขนาด 16x2
LiquidCrystal_I2C lcd(0x27, 16, 2);

// ====================================================
// 4. KEYPAD CONFIGURATION
// ====================================================


/*
 Keypad 4x4
 สายแรกเริ่มที่ D9 แล้วลดลงทีละขา

 R1 -> D9
 R2 -> D8
 R3 -> D7
 R4 -> D6

 C1 -> D5
 C2 -> D4
 C3 -> D3
 C4 -> D2
*/

// ขา Row
const byte ROW_PINS[4] = {
  9, 8, 7, 6
};

// ขา Column
const byte COL_PINS[4] = {
  5, 4, 3, 2
};

// ตำแหน่งปุ่ม
const char KEYS[4][4] = {
  {'1', '2', '3', 'A'},
  {'4', '5', '6', 'B'},
  {'7', '8', '9', 'C'},
  {'*', '0', '#', 'D'}
};


// ====================================================
// 5. SYSTEM VARIABLES
// ====================================================

// เก็บปุ่มที่กดล่าสุด
char lastKey = '-';

// เก็บปุ่มที่อ่านได้ล่าสุดจากวงจร
char lastRawKey = 0;

// เก็บปุ่มที่ผ่าน Debounce แล้ว
char stableKey = 0;

// เวลาที่ค่าปุ่มเปลี่ยน
unsigned long lastKeyChange = 0;

// เวลากรองการเด้งของปุ่ม
const unsigned long DEBOUNCE_TIME = 40;

// เวลาที่พยายามเชื่อมต่อล่าสุด
unsigned long lastRetry = 0;

// ระยะเวลาลองเชื่อมต่อใหม่
const unsigned long RETRY_INTERVAL = 5000;

// เวลาที่อัปเดตสถานะ LCD ล่าสุด
unsigned long lastLCDUpdate = 0;

// อัปเดตสถานะ LCD ทุก 1 วินาที
const unsigned long LCD_INTERVAL = 1000;

// ====================================================
// 6. LCD FUNCTION
// ====================================================

// อัปเดตหน้าจอ LCD
void updateLCD() {

  // บรรทัดแรก แสดงปุ่มล่าสุด
  lcd.setCursor(0, 0);

  lcd.print("Last Key: ");
  lcd.print(lastKey);
  lcd.print("     ");

  // บรรทัดที่สอง แสดงสถานะการเชื่อมต่อ
  lcd.setCursor(0, 1);

  if (WiFi.status() != WL_CONNECTED) {

    lcd.print("WiFi: OFF       ");

  } else if (!mqtt.connected()) {

    lcd.print("MQTT: OFF       ");

  } else {

    lcd.print("MQTT: ONLINE    ");
  }
}

// ====================================================
// 7. KEYPAD SCANNING FUNCTION
// ====================================================

// อ่าน Keypad โดยไม่ใช้ Keypad.h
char scanKeypad() {

  // ตั้ง Row ทุกแถวเป็น HIGH ก่อน
  for (int row = 0; row < 4; row++) {
    digitalWrite(ROW_PINS[row], HIGH);
  }

  // สแกน Row ทีละแถว
  for (int row = 0; row < 4; row++) {

    // ดึง Row ปัจจุบันเป็น LOW
    digitalWrite(ROW_PINS[row], LOW);

    // รอสัญญาณนิ่ง
    delayMicroseconds(10);

    // อ่าน Column ทั้ง 4
    for (int col = 0; col < 4; col++) {

      // เมื่อกดปุ่ม Column จะเป็น LOW
      if (digitalRead(COL_PINS[col]) == LOW) {

        // คืนค่า Row เป็น HIGH
        digitalWrite(ROW_PINS[row], HIGH);

        // คืนตัวอักษรปุ่มที่กด
        return KEYS[row][col];
      }
    }

    // คืนค่า Row เป็น HIGH ก่อนสแกนแถวต่อไป
    digitalWrite(ROW_PINS[row], HIGH);
  }

  // ไม่มีปุ่มถูกกด
  return 0;
}

// ====================================================
// 8. KEYPAD DEBOUNCE FUNCTION
// ====================================================

// อ่านและตรวจจับการกดปุ่ม
void updateKeypad() {

  // อ่านปุ่มจาก Keypad
  char rawKey = scanKeypad();

  // ถ้าปุ่มเปลี่ยน เริ่มจับเวลาใหม่
  if (rawKey != lastRawKey) {
    lastKeyChange = millis();
  }

  // เมื่อสถานะนิ่งครบเวลาที่กำหนด
  if (millis() - lastKeyChange >= DEBOUNCE_TIME) {

    // ตรวจสอบว่าสถานะเปลี่ยนจริง
    if (rawKey != stableKey) {

      // เก็บสถานะใหม่
      stableKey = rawKey;

      // ส่งเฉพาะตอนมีการกดปุ่ม
      if (stableKey != 0) {

        // บันทึกปุ่มล่าสุด
        lastKey = stableKey;

        // แสดงปุ่มผ่าน Serial Monitor
        Serial.print("Key Pressed: ");
        Serial.println(lastKey);

        // อัปเดต LCD ทันที
        updateLCD();

        // ส่งข้อมูลผ่าน MQTT
        if (mqtt.connected()) {

          // แปลงปุ่มเป็นข้อความ
          String payload = String(lastKey);

          // Publish ไปยัง Topic
          if (mqtt.publish(MQTT_TOPIC, payload)) {

            Serial.print("MQTT Published: ");
            Serial.println(payload);

          } else {

            Serial.println("MQTT Publish Failed!");
          }

        } else {

          Serial.println("MQTT Offline!");
        }
      }
    }
  }

  // เก็บค่าที่อ่านล่าสุด
  lastRawKey = rawKey;
}

// ====================================================
// 9. WIFI FUNCTION
// ====================================================

// เชื่อมต่อ Wi-Fi
void connectWiFi() {

  // ถ้าเชื่อมต่ออยู่แล้วไม่ต้องทำซ้ำ
  if (WiFi.status() == WL_CONNECTED) {
    return;
  }

  Serial.println("Connecting WiFi...");

  // เริ่มเชื่อมต่อ
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  // บันทึกเวลาเริ่มต้น
  unsigned long startTime = millis();

  // รอสูงสุด 10 วินาที
  while (WiFi.status() != WL_CONNECTED &&
         millis() - startTime < 10000) {
    delay(250);
  }

  // ตรวจสอบผล
  if (WiFi.status() == WL_CONNECTED) {

    Serial.println("WiFi Connected!");

    Serial.print("IP: ");
    Serial.println(WiFi.localIP());

  } else {

    Serial.println("WiFi Failed!");
  }
}

// ====================================================
// 10. MQTT FUNCTION
// ====================================================

// เชื่อมต่อ MQTT
void connectMQTT() {

  // ต้องมี Wi-Fi ก่อน
  if (WiFi.status() != WL_CONNECTED) {
    return;
  }

  // ถ้าเชื่อมต่อแล้วไม่ต้องทำซ้ำ
  if (mqtt.connected()) {
    return;
  }

  Serial.println("Connecting MQTT...");

  // สร้าง Client ID
  String clientId = "UNO-R4-KEYPAD-";
  clientId += String(millis(), HEX);

  // เชื่อมต่อ Broker
  if (mqtt.connect(clientId.c_str())) {

    Serial.println("MQTT Connected!");

  } else {

    Serial.println("MQTT Failed!");
  }
}

// ====================================================
// 11. SETUP
// ====================================================

void setup() {

  // เปิด Serial Monitor
  Serial.begin(9600);

  // กำหนด Row เป็น OUTPUT
  for (int row = 0; row < 4; row++) {

    pinMode(ROW_PINS[row], OUTPUT);

    // เริ่มต้นเป็น HIGH
    digitalWrite(ROW_PINS[row], HIGH);
  }

  // กำหนด Column เป็น INPUT_PULLUP
  for (int col = 0; col < 4; col++) {
    pinMode(COL_PINS[col], INPUT_PULLUP);
  }

  // เริ่ม I2C
  Wire.begin();

  // เริ่ม LCD
  lcd.init();

  // เปิด Backlight
  lcd.backlight();

  // ข้อความเริ่มต้น
  lcd.setCursor(0, 0);
  lcd.print("KEYPAD MQTT LCD ");

  lcd.setCursor(0, 1);
  lcd.print("Starting...     ");

  // เริ่ม MQTT
  mqtt.begin(MQTT_BROKER, MQTT_PORT, network);

  // จำกัดเวลารอ MQTT
  mqtt.setTimeout(1000);

  Serial.println("========================");
  Serial.println("KEYPAD MQTT LCD");
  Serial.println("========================");

  // เชื่อมต่อ Wi-Fi
  connectWiFi();

  // เชื่อมต่อ MQTT
  connectMQTT();

  // ล้าง LCD
  lcd.clear();

  // แสดงสถานะเริ่มต้น
  updateLCD();
}

// ====================================================
// 12. MAIN LOOP
// ====================================================

void loop() {

  // อ่าน Keypad ตลอดเวลา
  updateKeypad();

  // ดูแลการเชื่อมต่อ MQTT
  if (mqtt.connected()) {
    mqtt.loop();
  }

  // ตรวจสอบ Wi-Fi/MQTT ทุก 5 วินาที
  if (millis() - lastRetry >= RETRY_INTERVAL) {

    lastRetry = millis();

    // เชื่อมต่อ Wi-Fi ใหม่
    if (WiFi.status() != WL_CONNECTED) {
      connectWiFi();
    }

    // เชื่อมต่อ MQTT ใหม่
    if (WiFi.status() == WL_CONNECTED &&
        !mqtt.connected()) {
      connectMQTT();
    }
  }

  // อัปเดตสถานะ LCD ทุก 1 วินาที
  if (millis() - lastLCDUpdate >= LCD_INTERVAL) {

    lastLCDUpdate = millis();

    updateLCD();
  }
}
