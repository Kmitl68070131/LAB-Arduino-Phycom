/*
ชื่อ: MQTT LED Fading รับค่า Analog
อุปกรณ์: LED PWM
การต่อสาย: LED anode ผ่าน R → D9, cathode → GND
WiFi SSID: EDOK | รหัส WiFi อยู่ใน WIFI_PASS แก้ได้ที่ตัวแปรนี้
MQTT Broker: phycom.it.kmitl.ac.th:1883
MQTT Topic: 68070131/09_mqtt_pot_tx
แก้ขา: ตัวแปรส่วน LED PWM
แก้ความถี่: INTERVAL (ms)
Arduino UNO R4 WiFi; Serial Monitor 115200
ไลบรารี: WiFiS3, MQTTClient by 256dpi
วิธีใช้: คัดลอก .txt ลง Arduino IDE -> ตั้ง WiFi -> Upload -> เปิดเว็บ Dashboard
ข้อควรระวัง: โค้ดแต่ละไฟล์เป็น Sketch แยก; ตรวจ COM ของ 7-Segment และลำดับ Keypad
*/
#include <WiFiS3.h>
#include <MQTTClient.h>

const char WIFI_SSID[]="EDOK"; // ชื่อ WiFi
const char WIFI_PASS[]="1234567890"; // แก้รหัส WiFi ตรงนี้
const char MQTT_HOST[]="phycom.it.kmitl.ac.th"; // Broker
const char MQTT_TOPIC[]="68070131/09_mqtt_pot_tx"; // Topic เฉพาะข้อนี้
WiFiClient net;
MQTTClient mqtt(256);
const int LED=9;
// รับค่าทุกครั้งที่ MQTT ส่งเข้ามา ไม่ต้องตั้ง INTERVAL
unsigned long lastRetry=0;
void onMessage(String &topic,String &payload){
  // ข้อมูลมาจากเว็บผ่าน MQTT Subscribe
  analogWrite(LED,map(constrain(payload.toInt(),0,1023),0,1023,0,255)); // ค่า 0-1023 เป็น PWM 0-255
}
void setup(){
 Serial.begin(115200);
 pinMode(LED,OUTPUT);
 mqtt.begin(MQTT_HOST,1883,net);
 mqtt.onMessage(onMessage);
 WiFi.begin(WIFI_SSID,WIFI_PASS);
}
void loop(){
 if(WiFi.status()!=WL_CONNECTED){
  if(millis()-lastRetry>=10000UL){lastRetry=millis();WiFi.begin(WIFI_SSID,WIFI_PASS);}
  return;
 }
 if(!mqtt.connected()){
  if(millis()-lastRetry>=5000UL){
   lastRetry=millis();
   String clientId="genz-10_mqtt_led_fade_rx-"+String(millis(),HEX);
   if(mqtt.connect(clientId.c_str()))mqtt.subscribe(MQTT_TOPIC);
  }
  return;
 }
 mqtt.loop();
 // ตัวรับไม่ Publish กลับไป Topic เดิม เพื่อไม่ให้เกิด MQTT loop
}
