/*
ชื่อ: LCD Custom Font หัวใจ
อุปกรณ์: LCD I2C
การต่อสาย: SDA/SCL → SDA/SCL; VCC → 5V; GND → GND
WiFi SSID: EDOK | รหัส WiFi อยู่ใน WIFI_PASS แก้ได้ที่ตัวแปรนี้
MQTT Broker: phycom.it.kmitl.ac.th:1883
MQTT Topic: 68070131/05_lcd_heart
แก้ขา: ตัวแปรส่วน LCD I2C
แก้ความถี่: INTERVAL (ms)
Arduino UNO R4 WiFi; Serial Monitor 115200
ไลบรารี: WiFiS3, MQTTClient by 256dpi, Wire, LiquidCrystal_I2C
วิธีใช้: คัดลอก .txt ลง Arduino IDE -> ตั้ง WiFi -> Upload -> เปิดเว็บ Dashboard
ข้อควรระวัง: โค้ดแต่ละไฟล์เป็น Sketch แยก; ตรวจ COM ของ 7-Segment และลำดับ Keypad
*/
#include <WiFiS3.h>
#include <MQTTClient.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
LiquidCrystal_I2C lcd(0x27,16,2);
const char WIFI_SSID[]="EDOK"; // ชื่อ WiFi
const char WIFI_PASS[]="1234567890"; // แก้รหัส WiFi ตรงนี้
const char MQTT_HOST[]="phycom.it.kmitl.ac.th"; // Broker
const char MQTT_TOPIC[]="68070131/05_lcd_heart"; // Topic เฉพาะข้อนี้
WiFiClient net;
MQTTClient mqtt(256);
byte heart[8]={0,10,31,31,14,4,0,0};int counter=0;
const unsigned long INTERVAL=1000UL; // ช่วงเวลาการทำงาน
unsigned long lastRead=0,lastRetry=0;
void setup(){
 Serial.begin(115200);
 Wire.begin();lcd.init();lcd.backlight();lcd.createChar(0,heart);
 mqtt.begin(MQTT_HOST,1883,net);
 
 WiFi.begin(WIFI_SSID,WIFI_PASS);
}
void loop(){
 if(WiFi.status()!=WL_CONNECTED){
  if(millis()-lastRetry>=10000UL){lastRetry=millis();WiFi.begin(WIFI_SSID,WIFI_PASS);}
  return;
 }
 if(!mqtt.connected()){
  if(millis()-lastRetry>=5000UL){
   lastRetry=millis();
   String clientId="genz-05_lcd_heart-"+String(millis(),HEX);
   if(mqtt.connect(clientId.c_str())){}
  }
  return;
 }
 mqtt.loop();
 if(millis()-lastRead<INTERVAL)return;
 lastRead=millis();
 // ส่วนการอ่าน Sensor / การแสดงผล แก้ได้ตรงนี้
 lcd.setCursor(0,0);lcd.print("I ");lcd.write((uint8_t)0);lcd.print(" IT         ");lcd.setCursor(0,1);lcd.print("68070131 ");lcd.print(counter++);String payload="I HEART IT";
 Serial.println(payload);
 
 mqtt.publish(MQTT_TOPIC,payload);
}
