/*
  GEN_Z / Potentiometer | Arduino UNO R4 WiFi
  หน้าที่: อ่าน ADC 12 บิต 0–4095
  การต่อสาย: ขานอก -> 5V และ GND; ขากลาง -> A0
  WiFi: EDOK / 1234567890 (แก้ WIFI_SSID และ WIFI_PASS)
  MQTT: phycom.it.kmitl.ac.th:1883 (แก้ MQTT_HOST และ MQTT_PORT)
  Topic: 68070131/pot (แก้ TOPIC)
  เปิด Serial Monitor 115200 baud
  ใช้ไลบรารี WiFiS3 และ MQTTClient by 256dpi
  ไฟล์นี้เป็น Sketch เดี่ยว: คัดลอกลง Arduino IDE และบันทึกเป็น .ino ได้
  เว็บ: GEN_Z/WEB/dashboard.html (ต้องเปิดผ่าน Bridge server.js)
  หมายเหตุ: Wi-Fi password นี้จะมองเห็นได้ใน GitHub สาธารณะ
*/
#include <WiFiS3.h>
#include <MQTTClient.h>

const char WIFI_SSID[]="EDOK"; // แก้ชื่อ Wi-Fi
const char WIFI_PASS[]="1234567890"; // แก้รหัส Wi-Fi
const char MQTT_HOST[]="phycom.it.kmitl.ac.th"; // แก้ Broker
const int MQTT_PORT=1883; // TCP MQTT
const char TOPIC[]="68070131/pot"; // แก้ Topic
WiFiClient net;
MQTTClient mqtt(256);
const int POT_PIN=A0;
unsigned long lastRead=0,lastRetry=0;
const unsigned long READ_INTERVAL=500UL; // รอบการอ่าน
void setup(){
  Serial.begin(115200);
  analogReadResolution(12);
  mqtt.begin(MQTT_HOST,MQTT_PORT,net);
  WiFi.begin(WIFI_SSID,WIFI_PASS);
}
void loop(){
  // ไม่บล็อกการอ่านเซนเซอร์ระหว่างรอ Wi-Fi
  if(WiFi.status()!=WL_CONNECTED){
    if(millis()-lastRetry>=10000UL){lastRetry=millis();WiFi.begin(WIFI_SSID,WIFI_PASS);}
    return;
  }
  if(!mqtt.connected()){
    if(millis()-lastRetry>=5000UL){
      lastRetry=millis();
      String clientId="GENZ-pot-"+String(millis(),HEX); // ID ไม่ชนง่าย
      mqtt.connect(clientId.c_str());
    }
    return;
  }
  mqtt.loop(); // ต้องเรียกต่อเนื่องเพื่อรักษาการเชื่อมต่อ
  if(millis()-lastRead<READ_INTERVAL)return;
  lastRead=millis();
  // ---- ส่วนอ่านอุปกรณ์: ปรับตรรกะได้ที่นี่ ----
  int raw=analogRead(POT_PIN);
  String payload=String(raw);
  Serial.print(TOPIC);Serial.print(" = ");Serial.println(payload);
  mqtt.publish(TOPIC,payload); // ส่งไปหน้าเว็บผ่าน Broker
}
