02 / สูตรคำนวณแยกตาม Sensor

TMP36 — อุณหภูมิ °C

ใช้ Analog Output: V = ADC × Vref / (2ⁿ − 1) และ °C = (V − 0.5) × 100 โดย TMP36 มี Offset 500 mV และ 10 mV/°C

analogReadResolution(12); // UNO R4: ADC 0–4095
int raw = analogRead(A3);
float voltage = raw * 5.0 / 4095.0; // 5.0 คือ Vref ที่ต้องตรวจวัดจริง
float celsius = (voltage - 0.5) * 100.0;
Serial.println(celsius, 2);
ตรวจการจ่ายไฟ, Vref, และ Pinout ของ TMP36 ก่อนใช้งาน; ค่าจริงขึ้นกับการคาลิเบรต

LM35 — อุณหภูมิ °C

LM35 มีอัตรา 10 mV/°C โดยไม่มี Offset 500 mV แบบ TMP36: °C = V × 100

analogReadResolution(12);
float voltage = analogRead(A0) * 5.0 / 4095.0;
float celsius = voltage * 100.0;

NTC Thermistor — อุณหภูมิ °C

ใช้ตัวต้านทานคงที่กับ NTC เป็น Voltage Divider จากนั้นใช้สมการ Beta: 1/T = 1/T₀ + ln(R/R₀)/β (T เป็น Kelvin)

#include <math.h>
analogReadResolution(12);
int adc = analogRead(A0);
if (adc > 0 && adc < 4095) {
  // ต่อ NTC ไป 5V และ R_FIXED ลง GND; วัดตรงจุดกลาง
  const float R_FIXED = 10000.0;
  float rNtc = R_FIXED * (4095.0 / adc - 1.0);
  float kelvin = 1.0 / (1.0/298.15 + log(rNtc/10000.0)/3950.0);
  float celsius = kelvin - 273.15;
}
ตัวอย่างนี้ใช้ NTC 10kΩ ที่ 25°C, β=3950 และตัวต้านทานคงที่ 10kΩ ต้องแก้ตาม datasheet และตำแหน่งการต่อวงจรจริง

DHT11 / DHT22 — อุณหภูมิและความชื้น

เซนเซอร์ส่งค่าที่คำนวณภายในมาแล้ว ไม่ต้องแปลงแรงดัน Analog

#include <DHT.h>
DHT dht(2, DHT22); // เปลี่ยน DHT22 เป็น DHT11 ตามรุ่น
void setup() { Serial.begin(115200); dht.begin(); }
void loop() {
  float t = dht.readTemperature();
  float h = dht.readHumidity();
  if (!isnan(t) && !isnan(h)) {
    Serial.print(t); Serial.print(" C / ");
    Serial.print(h); Serial.println(" %RH");
  }
  delay(2000);
}

HC-SR04 — ระยะทาง

ระยะทาง (cm) = เวลา Echo (µs) × 0.0343 / 2 โดย 0.0343 cm/µs เป็นความเร็วเสียงประมาณที่ 20°C

const int TRIG=2, ECHO=3;
void setup(){ pinMode(TRIG,OUTPUT); pinMode(ECHO,INPUT); }
void loop(){
 digitalWrite(TRIG,LOW); delayMicroseconds(2);
 digitalWrite(TRIG,HIGH); delayMicroseconds(10);
 digitalWrite(TRIG,LOW);
 unsigned long us=pulseIn(ECHO,HIGH,30000UL);
 if(us) {
   float cm=us*0.0343/2.0;
   // ถ้ามีอุณหภูมิจริง t °C:
   // cm = us * ((331.3 + 0.606*t)/10000.0) / 2.0;
 }
 delay(200);
}

Potentiometer / LDR — ระดับ Analog

แปลง ADC เป็นเปอร์เซ็นต์หรือ PWM: % = ADC × 100 / 4095 และ PWM = ADC × 255 / 4095 (เมื่อ ADC 12 บิต)

analogReadResolution(12);
int raw=analogRead(A0);
float percent=raw*100.0/4095.0;
int pwm=map(raw,0,4095,0,255);
analogWriteResolution(8);
analogWrite(9,pwm); // LED ผ่าน R 220–330Ω
LDR ไม่ได้วัดอุณหภูมิ; การแปลงเป็น Lux ต้องรู้วงจรแบ่งแรงดันและคาลิเบรตเซนเซอร์

PIR / Keypad / RGB / 7-Segment

อุปกรณ์กลุ่มนี้ไม่ใช่เซนเซอร์วัดอุณหภูมิ ใช้ digitalRead(), digitalWrite() หรือ analogWrite() เพื่ออ่านสถานะหรือควบคุมการแสดงผล

03 / ตัวอย่างคำสั่งที่ใช้บ่อย

คำสั่งใช้ทำอะไร
pinMode(2, INPUT_PULLUP)ตั้งปุ่มกดใช้ตัวต้านทาน Pull-up ภายใน
digitalRead(2)อ่านสถานะ HIGH / LOW
analogRead(A0)อ่านค่า Analog
analogWrite(9,128)สร้าง PWM ความสว่างประมาณครึ่งหนึ่งเมื่อใช้ 8 บิต
pulseIn(3,HIGH,30000)วัดเวลาพัลส์ Echo พร้อม Timeout
lcd.setCursor(0,1)ย้ายเคอร์เซอร์ไปต้นแถวที่สอง
mqtt.publish(topic,msg)ส่งข้อมูลไปยัง MQTT Broker
mqtt.subscribe(topic)สมัครรับข้อมูลจาก MQTT Broker
mqtt.loop()ประมวลผลข้อความ MQTT และรักษาการเชื่อมต่อ

04 / วิธีส่งข้อมูล Arduino ขึ้นเว็บไซต์ผ่าน MQTT

  1. ต่อเซนเซอร์กับ Arduino UNO R4 WiFi และติดตั้ง WiFiS3 / MQTTClient
  2. กำหนด Wi-Fi และ Broker phycom.it.kmitl.ac.th:1883 ในโค้ด Arduino
  3. Publish ค่าไปที่ Topic เช่น 68070131/tmp36
  4. ฝั่งเว็บต้อง Subscribe Topic เดียวกันผ่าน MQTT WebSocket หรือ Node.js Bridge
  5. เปิดหน้า MQTT Exercises เพื่อทดสอบการรับส่งข้อมูล (ต้องมี Bridge ที่เชื่อมต่อได้)

GitHub Pages เป็น Static Hosting: JavaScript บนเบราว์เซอร์เชื่อม TCP MQTT พอร์ต 1883 โดยตรงไม่ได้ ต้องมี WebSocket Endpoint หรือ Backend Bridge ที่ใช้งานจริง และควรจำกัดสิทธิ์การ Publish หากเปิดให้สาธารณะ

Arduino: ส่งอุณหภูมิ TMP36 ทุก 2 วินาที

คัดลอก Sketch นี้ไป Arduino IDE และเปลี่ยนรหัส Wi-Fi ในเครื่องของคุณ ไม่ควรเผยแพร่รหัสผ่านจริงใน Repository สาธารณะ

#include <WiFiS3.h>
#include <MQTTClient.h>

const char WIFI_SSID[] = "EDOK";
const char WIFI_PASS[] = "PUT_WIFI_PASSWORD_HERE";
const char BROKER[] = "phycom.it.kmitl.ac.th";
const char TOPIC[] = "68070131/tmp36";
WiFiClient network;
MQTTClient mqtt(256);
unsigned long lastRead = 0;
unsigned long lastReconnect = 0;
byte mac[6];

void setup() {
  Serial.begin(115200);
  analogReadResolution(12);
  WiFi.macAddress(mac);
  mqtt.begin(BROKER, 1883, network);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
}
void loop() {
  if (WiFi.status() != WL_CONNECTED) {
    if (millis()-lastReconnect > 10000UL) {
      lastReconnect=millis();
      WiFi.begin(WIFI_SSID,WIFI_PASS);
    }
    return;
  }
  if (!mqtt.connected()) {
    if (millis()-lastReconnect > 5000UL) {
      lastReconnect=millis();
      String id="R4-TMP-"+String(mac[4],HEX)+String(mac[5],HEX);
      mqtt.connect(id.c_str());
    }
    return;
  }
  mqtt.loop();
  if (millis()-lastRead >= 2000UL) {
    lastRead=millis();
    float voltage=analogRead(A3)*5.0/4095.0;
    float celsius=(voltage-0.5)*100.0;
    String message=String(celsius,2);
    mqtt.publish(TOPIC,message);
    Serial.println(message);
  }
}

Node.js Bridge: ส่ง MQTT ไปหน้าเว็บผ่าน Server-Sent Events

ติดตั้ง npm install express mqtt และใช้ Node.js ฝั่งเซิร์ฟเวอร์ที่เชื่อม Broker ได้ (ไม่ใช่ GitHub Pages)

const express = require('express');
const mqtt = require('mqtt');
const app = express();
const client = mqtt.connect('mqtt://phycom.it.kmitl.ac.th:1883');
const viewers = new Set();
app.use(express.static(__dirname));
app.get('/events', (req, res) => {
  res.setHeader('Content-Type','text/event-stream');
  res.setHeader('Cache-Control','no-cache');
  res.flushHeaders();
  viewers.add(res);
  req.on('close', () => viewers.delete(res));
});
client.on('connect', () => client.subscribe('68070131/#'));
client.on('message', (topic, value) => {
  const data = 'data: '+JSON.stringify({topic,value:value.toString()})+'\n\n';
  for (const res of viewers) res.write(data);
});
app.listen(3000, () => console.log('http://localhost:3000'));

  const msg = JSON.parse(event.data);
