รายการค่า
BoardArduino UNO R4 WiFi
Wi-FiEDOK (กำหนดรหัสผ่านใน Sketch)
MQTT Brokerphycom.it.kmitl.ac.th
Port1883 (MQTT over TCP)
Topic Prefix68070131/
ไลบรารีWiFiS3.h, MQTTClient.h โดย 256dpi

Publish กับ Subscribe ต่างกันอย่างไร

ทิศทางTopic ตัวอย่างการทำงาน
Arduino → เว็บ68070131/mqtt/dataบอร์ดอ่านค่าแล้ว mqtt.publish() ไปให้เว็บรับ
เว็บ → Arduino68070131/mqtt/setบอร์ด mqtt.subscribe() แล้วรับคำสั่งใน Callback
Arduino → เว็บ68070131/mqtt/statusส่ง ONLINE หรือสถานะตอบกลับ

สำคัญ: เปลี่ยนชื่อ Topic ตรงค่าคงที่ TOPIC_DATA, TOPIC_SET, TOPIC_ACK และต้องเปลี่ยนฝั่งเว็บให้ตรงกันทุกตัวอักษร

โค้ดตัวอย่างครบ: รับคำสั่ง LED และส่งข้อมูลขึ้นเว็บ

แสดงตำแหน่งแก้ Wi-Fi, Broker และ Topic พร้อมคอมเมนต์ทุกส่วน ตัวอย่างนี้ใช้ LED บนบอร์ด ไม่ต้องต่ออุปกรณ์ภายนอก

#include <WiFiS3.h>            // Wi-Fi ของ Arduino UNO R4 WiFi
#include <MQTTClient.h>        // ไลบรารี MQTT by 256dpi

const char SSID[] = "EDOK";    // <<< เปลี่ยนชื่อ Wi-Fi
const char PASS[] = "1234567890"; // <<< เปลี่ยนรหัส Wi-Fi
const char HOST[] = "phycom.it.kmitl.ac.th"; // <<< เปลี่ยน Broker
const int PORT = 1883;         // <<< เปลี่ยนพอร์ต MQTT TCP

const char TOPIC_DATA[] = "68070131/mqtt/data"; // <<< เปลี่ยน Topic ส่ง
const char TOPIC_SET[] = "68070131/mqtt/set";   // <<< เปลี่ยน Topic รับ
const char TOPIC_ACK[] = "68070131/mqtt/status"; // <<< เปลี่ยน Topic ตอบกลับ

WiFiClient network;            // ช่องทาง TCP ผ่าน Wi-Fi
MQTTClient mqtt(512);          // บัฟเฟอร์ข้อความ MQTT
unsigned long lastSent = 0;    // เวลา Publish ล่าสุด
unsigned long lastTry = 0;     // เวลาลอง Connect ครั้งล่าสุด
String received = "";          // ข้อความที่รับจากเว็บ

void onMessage(String &topic, String &payload) {
  if (topic != TOPIC_SET) return; // ตรวจว่าเป็น Topic รับคำสั่ง
  received = payload;             // เก็บค่าที่ได้รับ
  Serial.println("RX: " + received);
}

void setup() {
  Serial.begin(115200);           // เปิด Serial Monitor
  pinMode(LED_BUILTIN, OUTPUT);   // LED บนบอร์ดสำหรับทดสอบ
  WiFi.begin(SSID, PASS);         // เริ่มต่อ Wi-Fi
  mqtt.begin(HOST, PORT, network); // ตั้ง Broker และช่องทาง TCP
  mqtt.onMessage(onMessage);      // ตั้ง Callback สำหรับ Subscribe
}

void loop() {
  if (WiFi.status() != WL_CONNECTED) {
    if (millis() - lastTry > 10000UL) {
      lastTry = millis();
      WiFi.begin(SSID, PASS);     // เชื่อม Wi-Fi ใหม่เมื่อหลุด
    }
    return;
  }

  if (!mqtt.connected()) {
    if (millis() - lastTry > 5000UL) {
      lastTry = millis();
      if (mqtt.connect("uno-r4-68070131-example")) {
        mqtt.subscribe(TOPIC_SET); // <<< Subscribe Topic หลัง Connect
        mqtt.publish(TOPIC_ACK, "ONLINE"); // <<< ส่งสถานะ
      }
    }
    return;
  }

  mqtt.loop();                   // เรียกบ่อย ๆ เพื่อรับข้อความ MQTT
  if (received == "ON") digitalWrite(LED_BUILTIN, HIGH);
  if (received == "OFF") digitalWrite(LED_BUILTIN, LOW);

  if (millis() - lastSent >= 2000UL) {
    lastSent = millis();
    mqtt.publish(TOPIC_DATA, String(millis())); // <<< Publish ขึ้นเว็บ
  }
}

คัดลอกโค้ดนี้ลง Arduino IDE เป็น Sketch ใหม่ แล้วเลือกบอร์ด UNO R4 WiFi ตรวจไลบรารีและเปิด Serial Monitor 115200 ต้องมี Wi-Fi และ Broker ที่เข้าถึงได้จึงจะเห็นข้อมูลบนเว็บ

ข้อผิดพลาดที่พบบ่อย

  • Wi-Fi ไม่เชื่อม: ตรวจ SSID, Password, ความแรงสัญญาณ และว่าเครือข่ายรองรับบอร์ด
  • MQTT ไม่เชื่อม: ตรวจ Host, Port 1883, Firewall, Client ID ไม่ซ้ำ และการเข้าถึง Broker
  • Publish ได้ แต่เว็บไม่เห็น: ตรวจ Topic ให้ตรง, ดูว่าเว็บ Subscribe ถูกตัวหรือไม่ และเปิดตัวรับ MQTT จริง
  • Arduino ไม่รับคำสั่ง: ตรวจว่า mqtt.loop() ทำงานสม่ำเสมอ และ Subscribe ใหม่หลัง Reconnect
  • ตัวอย่างเว็บ GitHub Pages รับไม่ได้: Browser ไม่เปิด MQTT TCP 1883 โดยตรง ต้องใช้ MQTT over WebSocket หรือ Node.js Bridge ที่เปิดทำงานอยู่
  • LCD ไม่ขึ้น: ตรวจ SDA/SCL, VCC/GND, Address 0x27 หรือ 0x3F และปรับ Contrast

Broker พอร์ต 1883 ไม่ใช้ TLS และอาจเป็นบริการร่วมกัน อย่าส่งข้อมูลส่วนตัว รหัสผ่าน หรือคำสั่งสำคัญโดยไม่มีระบบยืนยันตัวตน
